# Deliberately minimal. Both sibling projects are stdlib-heavy by design and
# this one keeps that property: everything outside the API process and the
# database driver uses only the standard library.
fastapi
uvicorn[standard]
psycopg2-binary
pgvector

# DocuSign's JWT Bearer Grant (server/providers/docusign_esign.py) requires
# RS256-signing a JWT with an RSA private key. Hand-rolling RSA/PKCS1v15
# signing in stdlib is exactly the kind of thing that becomes a real security
# bug, so this one narrow, vetted dependency is used for the signature only --
# not a JWT/OAuth framework, just RSA signing.
cryptography

# Loads .env into the environment for local dev (server/config.py) --
# a real env var an operator set always wins; this only populates what
# isn't already there. Not a secrets store, just a convenience layer over
# os.environ, so it doesn't conflict with "secrets live in env, never in
# the database and never in a config file."
python-dotenv

# Development only.
pytest
httpx
